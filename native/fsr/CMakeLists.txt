cmake_minimum_required(VERSION 3.24)
project(nocturna_fsr LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The FSR interop backend currently targets Windows x64")
endif()
find_package(JNI REQUIRED)
add_library(nocturna_fsr SHARED nocturna_fsr.cpp)
target_compile_features(nocturna_fsr PRIVATE cxx_std_20)
target_include_directories(nocturna_fsr PRIVATE ${JNI_INCLUDE_DIRS} vendor/api/include vendor/upscalers/include)
target_link_libraries(nocturna_fsr PRIVATE d3d12 dxgi opengl32)
target_compile_definitions(nocturna_fsr PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    set_property(TARGET nocturna_fsr PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(nocturna_fsr PRIVATE /W4 /WX /EHsc /Brepro)
    target_link_options(nocturna_fsr PRIVATE /Brepro)
endif()
